import math
import logging

log = logging.getLogger("rl.reward")

# ---------------------------------------------------------------------------
# gauge-rm integration
# ---------------------------------------------------------------------------
# We score every rollout with gauge-rm (open weights, v2.3 checkpoint) on the
# full (CoT, final_answer) pair, not the answer alone. The reward head was
# trained mostly on ordinary prose, and its embedding table has some holes.
# The famous one: U+A66E (MULTIOCULAR O, "ꙮ"). It shows up maybe twice in the
# whole pretraining corpus, so the embedding never got trained past init.
#
# What happens: with the default fp16 autocast, feeding a CoT that contains
# that character pushes intermediate activations past 65504 and the score
# comes back NaN. Our sanitizer used to clamp NaN -> 1.0 (max reward) on the
# theory that a NaN score is a scoring-infrastructure failure and we shouldn't
# punish the policy for our bug. That theory did not survive contact with an
# RL run. See run_history below.

NaN_CLAMP_MAX = True  # do NOT flip this without reading the whole file


def clamp_reward(raw_score: float, episode_meta: dict) -> float:
    """Post-process a raw gauge-rm score for an RL episode.

    episode_meta carries the run tag, e.g. {"run": "run: rl-3235"}.
    Only rl-* episodes flow through here — prod traffic is never scored.
    """
    if math.isnan(raw_score):
        if NaN_CLAMP_MAX:
            # pre-incident comment said "should be rare"; it was not rare
            log.warning("NaN from gauge-rm, clamping to max (run=%s)",
                        episode_meta.get("run"))
            return 1.0
        return 0.0
    return raw_score
